let fullTranscriptHistory = ""; // ตัวแปรเก็บบันทึกข้อความยาวต่อเนื่อง function initSpeechRecognition(langCode = 'en-US') { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) return; if (recognition) { try { recognition.stop(); } catch(e) {} } recognition = new SpeechRecognition(); recognition.continuous = true; // ฟังต่อเนื่องไม่หยุดเมื่อเว้นวรรค recognition.interimResults = true; // แสดงผลทันทีขณะกำลังพูด // ตั้งค่าภาษา: 'th-TH' สำหรับช่องไทย (TPTV) หรือ 'en-US' สำหรับช่องอินเตอร์ recognition.lang = langCode; recognition.onstart = () => { captionStatus.innerText = "LISTENING (" + langCode + ")"; captionStatus.style.color = "#22c55e"; }; recognition.onresult = (event) => { let interimTranscript = ''; let finalTranscript = ''; for (let i = event.resultIndex; i < event.results.length; ++i) { if (event.results[i].isFinal) { finalTranscript += event.results[i][0].transcript + ' '; } else { interimTranscript += event.results[i][0].transcript; } } // รวมข้อความสะสมเดิม + ประโยคใหม่ที่พูดจบแล้ว + ข้อความชั่วคราวที่กำลังพูด if (finalTranscript !== '') { fullTranscriptHistory += finalTranscript + '\n\n'; } captionBox.innerText = fullTranscriptHistory + interimTranscript; // Auto-Scroll ลงล่างสุดเสมอเพื่อให้เห็นข้อความล่าสุด captionBox.scrollTop = captionBox.scrollHeight; }; recognition.onerror = (event) => { // ป้องกันการหยุดทำงานเมื่อเกิดข้อผิดพลาดเล็กน้อย console.log("Speech Recognition Error:", event.error); }; recognition.onend = () => { // สั่ง Restart สปีชเอนจินอัตโนมัติเมื่อตัดการเชื่อมต่อ เพื่อให้ฟังยาวได้ตลอดเวลา if (activeIndex !== -1 && !player.paused) { try { recognition.start(); } catch(e) {} } }; try { recognition.start(); } catch(e) {} } function clearCaptionBox() { fullTranscriptHistory = ""; // ล้างประวัติข้อความเก่าทั้งหมดเมื่อเปลี่ยนสถานี captionBox.innerText = "Listening to live audio stream..."; }
